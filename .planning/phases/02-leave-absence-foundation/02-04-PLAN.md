```typescript
const qc = useQueryClient();
const { data: absenceList = [] } = useQuery({
  queryKey: ["absences"],
  queryFn: () => absenceApi.list().then(r => r.data),
});
const { data: assistants = [] } = useQuery({
  queryKey: ["assistants"],
  queryFn: () => assistantsApi.list().then(r => r.data),
});
const createAbsence = useMutation({
  mutationFn: (data: Record<string, unknown>) => absenceApi.create(data),
  onSuccess: () => {
    qc.invalidateQueries({ queryKey: ["absences"] });
    // also invalidate per-assistant balances
    assistants.forEach(a => qc.invalidateQueries({ queryKey: ["absences","balance",a.id] }));
    setDialogOpen(false);
  },
});
const deleteAbsence = useMutation({
  mutationFn: (id: string) => absenceApi.delete(id),
  onSuccess: () => {
    qc.invalidateQueries({ queryKey: ["absences"] });
    assistants.forEach(a => qc.invalidateQueries({ queryKey: ["absences","balance",a.id] }));
    setConfirmDeleteId(null);
  },
});
```

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-02-04-01 | Tampering | Record absence form inputs | mitigate | Client-side Zod-equivalent validation (startDate <= endDate, required fields). Server also validates via Zod (Plan 02-03). Client validation is UX; server is authoritative. |
| T-02-04-02 | Information Disclosure | absenceApi.list response | mitigate | Response only contains absences for the authenticated guardian. Server enforces guardianId scoping. Client displays what server returns — no client-side filtering of other guardians' data needed. |
| T-02-04-03 | Tampering | Client-side filter state | accept | Filters operate on already-fetched data for this guardian only. A client-side tampering of filter state can only show/hide rows already fetched — no data leakage across guardians. |
  | T-02-04-04 | Information Disclosure | AssistantAvatar in balance cards | accept | Assistant names and IDs are already visible on the Assistants page. Showing them on the Frånvaro page adds no new disclosure. |
